cmake_minimum_required(VERSION 3.20)
project(RouteMind VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(routemind-core INTERFACE)
target_include_directories(routemind-core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/core)
add_executable(routemind-core-tests tests/core-tests.cc)
target_link_libraries(routemind-core-tests PRIVATE routemind-core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(routemind-core-tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME core-contract COMMAND routemind-core-tests)
